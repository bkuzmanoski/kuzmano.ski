```swift
import Foundation

/// An item as the store tracks it.
public struct Item: Identifiable, Sendable {
  public let id: UUID
  public var title: String
  public var frame: CGRect
  public var isPinned: Bool = false

  static let minimumSize = CGSize(width: 320, height: 240)
}

enum LoadState<Value> {
  case idle
  case loading(progress: Double)
  case loaded(Value)
  case failed(any Error)
}

protocol ItemStoring: AnyObject {
  var items: [Item] { get }
  func moveToEnd(_ id: UUID) throws
}

extension ItemStoring {
  func summary(for id: UUID) -> String {
    guard let item = items.first(where: { $0.id == id }) else {
      return "No item with ID \(id)."
    }

    let label = item.title.isEmpty ? "Untitled" : item.title

    switch (item.isPinned, item.frame.width) {
    case (true, _):
      return "\(label) is pinned."
    case (false, let width) where width < Item.minimumSize.width:
      return "\(label) is only \(Int(width))pt wide."
    default:
      return String(format: "%@ — %.0f×%.0f", label, item.frame.width, item.frame.height)
    }
  }
}

@MainActor
final class Store: ItemStoring {
  private(set) var items: [Item] = []

  func moveToEnd(_ id: UUID) throws {
    guard let index = items.firstIndex(where: { $0.id == id }) else {
      throw CocoaError(.fileNoSuchFile)
    }

    items.append(items.remove(at: index))
  }
}
```
